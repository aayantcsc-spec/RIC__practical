# Practical 1A: Create and save pie charts in the current R working directory
# (basic pie, pie with percentages + legend, 3D pie)

# ---- 1. Basic pie chart ----
x <- c(21, 62, 10, 53)
labels <- c("London", "New York", "Singapore", "Mumbai")
png(file = "city_title_colours.png")
pie(x, labels, main = "City pie chart", col = rainbow(length(x)))
dev.off()

# ---- 2. Pie chart with percentage slices and legend ----
piepercent <- round(100 * x / sum(x), 1)
png(file = "city_percentage_legends.png")
pie(x, labels = piepercent, main = "City Pie Chart", col = rainbow(length(x)))
legend("topright", c("London", "New York", "Singapore", "Mumbai"),
       cex = 0.8, fill = rainbow(length(x)))
dev.off()

# ---- 3. 3D pie chart (needs: install.packages("plotrix")) ----
library(plotrix)
lbl <- c("London", "New York", "Singapore", "Mumbai")
png(file = "3d_pie_chart.png")
pie3D(x, labels = lbl, explode = 0.1, main = "Pie Chart of Countries")
dev.off()
