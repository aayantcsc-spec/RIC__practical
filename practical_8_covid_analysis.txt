rm(list = ls())
data1 <- read.csv("covid_state.csv")
print(head(data1))

# ---- helper functions (defined once, used by every question) ----
sumfun <- function(y) {
  sum <- 0
  for (i in 1:(length(y))) {
    sum <- sum + y[i]
  }
  return(sum)
}

findindex <- function(v, a) {
  val <- 0
  for (i in 1:(length(a))) {
    if (a[i] == v) {
      val <- i
      break
    }
  }
  return(val)          # FIX: journal returned i (always the last index) -> wrong "best" state
}

# returns rows of one state between two dates
get_state <- function(st, from, to) {
  subset(data1, State == st & as.Date(c_date) >= as.Date(from) & as.Date(c_date) <= as.Date(to))
}

# ===== Q1. Compare total covid cases of April for 5 states; best recovery percentage =====
mh <- get_state("MH", "2020-04-01", "2020-04-30")
ap <- get_state("AP", "2020-04-01", "2020-04-30")
ar <- get_state("AR", "2020-04-01", "2020-04-30")
as <- get_state("AS", "2020-04-01", "2020-04-30")
br <- get_state("BR", "2020-04-01", "2020-04-30")

rec_mh <- sumfun(mh$Recovered); con_mh <- sumfun(mh$tot_cases)
rec_ap <- sumfun(ap$Recovered); con_ap <- sumfun(ap$tot_cases)
rec_ar <- sumfun(ar$Recovered); con_ar <- sumfun(ar$tot_cases)
rec_as <- sumfun(as$Recovered); con_as <- sumfun(as$tot_cases)
rec_br <- sumfun(br$Recovered); con_br <- sumfun(br$tot_cases)

mh_rec_per <- (rec_mh / con_mh) * 100
ap_rec_per <- (rec_ap / con_ap) * 100
ar_rec_per <- (rec_ar / con_ar) * 100
as_rec_per <- (rec_as / con_as) * 100
br_rec_per <- (rec_br / con_br) * 100

labels1 <- c("Maharashtra", "Andhra Pradesh", "Arunachal Pradesh", "Assam", "Bihar")
perc_list <- c(mh_rec_per, ap_rec_per, ar_rec_per, as_rec_per, br_rec_per)
print(perc_list)
max_perc_list <- max(perc_list)
labelindex <- findindex(max_perc_list, perc_list)
state1 <- labels1[labelindex]
cat(state1, " is the best state with a recovery rate of ", max_perc_list, ".\n")

x <- round(perc_list, 2)
png(file = "State_Recovery_Pie_Chart_1.png")
pie(x, labels = x, main = "State Recovery Pie Chart", col = rainbow(length(x)))
legend("topright", labels1, cex = 0.6, fill = rainbow(length(x)))
dev.off()

# ===== Q4. Month where maximum patients recovered, and test ratio (Maharashtra) =====
mh_mon <- subset(data1, State == "MH")
month_key <- format(as.Date(mh_mon$c_date), "%Y-%m")     # FIX: whole months (journal stopped on the 30th)
months_k <- c("2020-04", "2020-05", "2020-06", "2020-07", "2020-08", "2020-09", "2020-10")
labels2  <- c("April", "May", "June", "July", "August", "September", "October")
monthrec <- numeric(0)
testratio <- numeric(0)
for (m in months_k) {
  d <- mh_mon[month_key == m, ]
  rec_m <- sumfun(d$Recovered)
  tot_m <- sumfun(d$tot_tests)
  monthrec  <- c(monthrec, rec_m)
  testratio <- c(testratio, rec_m / tot_m)    # FIX: journal divided July by June's tests (totjun)
}
print(monthrec)
maxrecmonth <- max(monthrec)
print(maxrecmonth)
labelindex2 <- findindex(maxrecmonth, monthrec)
state3 <- labels2[labelindex2]
cat(state3, " is the best month with ", maxrecmonth, " recovered patients.\n")

testratioframe <- data.frame(months = labels2, test_ratio = testratio, stringsAsFactors = FALSE)
print("Test Ratio DataFrame=")
print(testratioframe)

png(file = "Test_Ratio_Histogram.png")
hist(testratioframe$test_ratio, xlab = "Test ratio (recovered / tests)", col = "aquamarine",
     border = "blue4", breaks = 5, main = "Histogram of test ratio")   # FIX: removed fixed xlim/ylim that clipped data
dev.off()
